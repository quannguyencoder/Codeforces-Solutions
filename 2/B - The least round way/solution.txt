def count_factors(x, factor):
    if x == 0:
        return float('inf')
        
    count = 0
    while x % factor == 0:
        count += 1
        x //= factor
    return count
 
def solve():
    n = int(input())
    matrix = []
    zero_pos = None
    
    for i in range(n):
        row = list(map(int, input().split()))
        for j in range(n):
            if row[j] == 0:
                zero_pos = (i, j)
        matrix.append(row)
        
    def get_dp(factor):
        dp = [[float('inf')] * n for _ in range(n)]
        dp[0][0] = count_factors(matrix[0][0], factor)
            
        for i in range(n):
            for j in range(n):
                if i == 0 and j == 0:
                    continue
                    
                cost = count_factors(matrix[i][j], factor)
                up = dp[i - 1][j] if i > 0 else float('inf')
                left = dp[i][j - 1] if j > 0 else float('inf')
                
                dp[i][j] = min(up, left) + cost
                
        return dp
 
    dp2 = get_dp(2)
    dp5 = get_dp(5)
    
    ans = min(dp2[n - 1][n - 1], dp5[n - 1][n - 1])
    if zero_pos is not None and ans > 1:
        print(1)
        r, c = zero_pos
        path = 'D' * r + 'R' * c + 'D' * (n - 1 - r) + 'R' * (n - 1 - c)
        print(path)
        return
        
    print(ans)
    
    best_dp = dp2 if dp2[n - 1][n - 1] < dp5[n - 1][n - 1] else dp5
    factor = 2 if best_dp is dp2 else 5
    
    path = []
    i = j = n - 1
    
    while i > 0 or j > 0:
        cost = count_factors(matrix[i][j], factor)
        if i > 0 and best_dp[i][j] == best_dp[i - 1][j] + cost:
            path.append('D')
            i -= 1
        else:
            path.append('R')
            j -= 1
            
    print(''.join(path[::-1]))
 
solve()
